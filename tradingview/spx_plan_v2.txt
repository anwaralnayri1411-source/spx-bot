// ------------------------------- دوال مساعدة --------------------------------
f_ncdf(x) =>
    t = 1.0 / (1.0 + 0.2316419 * math.abs(x))
    d = 0.3989423 * math.exp(-x * x / 2.0)
    p = d * t * (0.3193815 + t * (-0.3565638 + t * (1.781478 + t * (-1.821256 + t * 1.330274))))
    x > 0 ? 1.0 - p : p

f_bs(S, K, T, iv, isCall) =>
    float res = na
    if T > 0 and iv > 0 and S > 0 and K > 0
        d1 = (math.log(S / K) + (0.04 + 0.5 * iv * iv) * T) / (iv * math.sqrt(T))
        d2 = d1 - iv * math.sqrt(T)
        disc = math.exp(-0.04 * T)
        res := isCall ? S * f_ncdf(d1) - K * disc * f_ncdf(d2) : K * disc * f_ncdf(-d2) - S * f_ncdf(-d1)
    res

f_delta(S, K, T, iv, isCall) =>
    float res = na
    if T > 0 and iv > 0 and S > 0 and K > 0
        d1 = (math.log(S / K) + (0.04 + 0.5 * iv * iv) * T) / (iv * math.sqrt(T))
        res := isCall ? f_ncdf(d1) : f_ncdf(-d1)
    res

// أقرب مستوى فوق المرجع بفارق لا يقل عن gap
f_above(arr, ref, gap) =>
    float best = na
    for i = 0 to array.size(arr) - 1
        v = array.get(arr, i)
        if not na(v) and v > ref + gap
            if na(best) or v < best
                best := v
    best

// أقرب مستوى تحت المرجع بفارق لا يقل عن gap
f_below(arr, ref, gap) =>
    float best = na
    for i = 0 to array.size(arr) - 1
        v = array.get(arr, i)
        if not na(v) and v < ref - gap
            if na(best) or v > best
                best := v
    best

// أول سترايك OTM (من الأقرب للسعر إلى الأبعد) يقع سعره بين الحدين = أعلى دلتا ضمن الميزانية
f_pick(S, T, iv, isCall, step, pMin, pMax) =>
    float bestK = na
    float bestP = na
    base = math.round(S / step) * step
    for i = 0 to 60
        k = isCall ? base + i * step : base - i * step
        otm = isCall ? k > S : k < S
        if otm and na(bestK)
            p = f_bs(S, k, T, iv, isCall) * 100
            if not na(p) and p >= pMin and p <= pMax
                bestK := k
                bestP := p
    [bestK, bestP]

// مستوى الأصل الذي يعطي سعر العقد = target (تقسيم ثنائي)
f_solve(K, T, iv, isCall, target, lo0, hi0) =>
    sg = isCall ? 1.0 : -1.0
    float lo = lo0
    float hi = hi0
    float res = na
    glo = sg * (f_bs(lo, K, T, iv, isCall) - target)
    ghi = sg * (f_bs(hi, K, T, iv, isCall) - target)
    if glo < 0 and ghi > 0
        for i = 0 to 39
            mid = (lo + hi) / 2
            gm = sg * (f_bs(mid, K, T, iv, isCall) - target)
            if gm < 0
                lo := mid
            else
                hi := mid
        res := (lo + hi) / 2
    res
